async function supabaseRequest(path, options = {}) { if ( SUPABASE_URL.includes('COLE_AQUI') || SUPABASE_KEY.includes('COLE_AQUI') ) { throw new Error('Supabase ainda não foi configurado.'); } const response = await fetch( `${SUPABASE_URL}/rest/v1/${path}`, { ...options, headers: { 'apikey': SUPABASE_KEY, 'Authorization': `Bearer ${SUPABASE_KEY}`, 'Content-Type': 'application/json', ...(options.headers || {}) } } ); const text = await response.text(); if (!response.ok) { throw new Error( text || `Erro HTTP ${response.status}` ); } if (!text.trim()) { return null; } try { return JSON.parse(text); } catch { return text; } }
